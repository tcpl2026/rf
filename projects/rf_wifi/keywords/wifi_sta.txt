wpa_cli wep
    [Arguments]        ${interface}    ${ssid}    ${authmode}    ${wepmode}    ${wepindex}

    ${index}=          Evaluate        ${wepindex} - 1
    Execute Command    sudo ${wpa_cli} -i ${interface} remove_network all
    Execute Command    sudo ${wpa_cli} -i ${interface} add_network
    Execute Command    sudo ${wpa_cli} -i ${interface} set_network 0 ssid \\"${ssid}\\"
    Execute Command    sudo ${wpa_cli} -i ${interface} set_network 0 key_mgmt NONE
    IF                 '${wepmode}' == 'wep64'
                       Execute Command    sudo ${wpa_cli} -i ${interface} set_network 0 wep_key${index} ${wep64key}
    ELSE IF            '${wepmode}' == 'wep128'
                       Execute Command    sudo ${wpa_cli} -i ${interface} set_network 0 wep_key${index} ${wep128key}
    END

    Execute Command    sudo ${wpa_cli} -i ${interface} set_network 0 wep_tx_keyidx ${index}
    IF                 '${authmode}' == 'open'
                       Execute Command    sudo ${wpa_cli} -i ${interface} set_network 0 auth_alg OPEN
    ELSE IF            '${authmode}' == 'shared'
                       Execute Command    sudo ${wpa_cli} -i ${interface} set_network 0 auth_alg SHARED
    END
    Execute Command    sudo ${wpa_cli} -i ${interface} enable_network 0

get ip
    [Arguments]            ${interface}
    Execute Command        sudo dhclient -r ${interface} 
    Execute Command        sudo dhclient ${interface} 
    ${output}=             Execute Command                  ifconfig ${interface}
    ${ip_list}=            Get Regexp Matches               ${output}                (?<=inet )(?:\\d{1,3}\\.){3}\\d{1,3}
    Log                    ${ip_list}
    Should Not Be Empty    ${ip_list}
    RETURN                 ${ip_list}[0]
