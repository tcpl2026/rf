ssh login
    [Arguments]        ${host}    ${user}        ${password}    ${alias}=default
    Open Connection    ${host}    alias=${alias}
    Login              ${user}    ${password}

start wpa_supplicant
    [Arguments]        ${interface}
    Execute Command    sudo pkill -9 wpa_supplicant
    Execute Command    sudo pkill -9 hostapd
    Execute Command    sudo ${wpa_supplicant} -i ${interface} -C /var/run/wpa_supplicant -B

open staut wpa_cli shell
    ssh login               ${staut_control_ip}    ${staut_user}    ${staut_pwd}    ${staut_wpa_cli}
    start wpa_supplicant    ${staut_wifi_interface}
    Sleep                   1s
    Write                   sudo ${wpa_cli} -i ${staut_wifi_interface}
