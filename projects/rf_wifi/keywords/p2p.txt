*** Keywords ***
Get Go MAC
    [Arguments]                    ${input}
    ${ret}=                        Get Lines Containing String       ${input}          go_dev_addr
    Log                            ${ret}
    ${ret}=                        Split String                      ${ret}            go_dev_addr=
    ${mac}=                        Get From List	             ${ret}            1
    RETURN                         ${mac}
    
Get P2P Interface
    [Arguments]                    ${interface}
    ${output}=                     Execute Command                   sudo ${wpa_cli} -i ${interface} interface
    Log                            ${output}
    ${p2p_interface}=              Get Lines Containing String       ${output}          p2p-${interface}-
    Log                            ${p2p_interface}
    Should Not Be Empty            ${p2p_interface}
    RETURN                         ${p2p_interface}

    Log                            ${output}

    ${output}=                     Read                          delay=0.5s
    Log                            ${output}
    Should Not Contain             ${output}                     p2p-${interface}-

Start P2P Go
    [Arguments]                    ${start_go_command}
    staut wpa_cli shell
    Write                          interface p2p-dev-${staut_wifi_interface}
    Sleep                          0.1s
    Write                          ${start_go_command}
    ${output}=                     Read                              delay=1s
    Log                            ${output}
    Should Contain                 ${output}                         OK
    ${p2p_go_mac}=                 Get Go MAC                        ${output}
    Log                            ${p2p_go_mac}
    RETURN                         ${p2p_go_mac}

Start P2P Gc
    [Arguments]                    ${go_mac}
    sta1 wpa_cli shell
    Write                          interface p2p-dev-${sta1_wifi_interface}
    Write                          p2p_connect ${go_mac} pbc join

Start P2P Gc PIN
    [Arguments]                    ${go_mac}
    sta1 wpa_cli shell
    Write                          interface p2p-dev-${sta1_wifi_interface}
    Sleep                          0.1s
    Write                          p2p_connect ${go_mac} pin join
    ${output}=                     Read                                delay=0.5s
    Log                            ${output}
    ${wps_pin}=                    Get Lines Matching Regexp           ${output}                          \\d{8}
    Log                            ${wps_pin}
    RETURN                         ${wps_pin}
                              

Check Go Success
    staut wpa_cli shell
    Set Client Configuration       timeout=10 seconds
    ${output}=                     Read Until                        EAPOL-4WAY-HS-COMPLETED
    Log                            ${output}
    Write                          interface ${staut_wifi_interface}

Check Gc Success
    sta1 wpa_cli shell
    Set Client Configuration       timeout=10 seconds
    ${output}=                     Read Until                                     P2P-GROUP-FORMATION-SUCCESS
    Log                            ${output}
    Write                          interface ${sta1_wifi_interface}

P2P PBC Autonomous Go and Go Start PBC First
    [Documentation]                Test steps:
    ...                            start autonomous go and start wps pbc
    ...                            then start wps pbc on gc

    [Arguments]                    ${go_command}
    # start p2p go and get go mac
    ${p2p_go_mac}=                 Start P2P GO                      ${go_command}

    # get go interface
    ${p2p_go_interface}=           staut get p2p interface
    Log                            ${p2p_go_interface}

    # start go wps_pbc
    staut wpa_cli shell
    Write                          interface ${p2p_go_interface}
    Sleep                          0.1s
    Write                          status
    Sleep                          0.1s
    Write                          wps_pbc
    ${output}=                     Read                              delay=0.5s
    Log                            ${output}

    # start gc wps_pbc
    Start P2P Gc                   ${p2p_go_mac}
    Check Gc Success

    # get gc interface
    ${p2p_gc_interface}=           sta1 get p2p interface
    Log                            ${p2p_gc_interface}

    Check Go Success

    Ping and Run Iperf             ${p2p_go_interface}               ${p2p_gc_interface}

P2P PBC Autonomous Go Enter Gc PIN In Go
    [Documentation]                Test steps:
    ...                            start autonomous go
    ...                            start wps pin on gc
    ...                            enter gc pin on go

    [Arguments]                    ${go_command}
    # start p2p go and get go mac
    ${p2p_go_mac}=                 Start P2P GO                      ${go_command}

    # start gc wps_pbc
    ${gc_pin}=                     Start P2P Gc PIN                  ${p2p_go_mac}

    # get go interface
    ${p2p_go_interface}=           staut get p2p interface
    Log                            ${p2p_go_interface}

    # enter gc pin on go
    staut wpa_cli shell
    Write                          interface ${p2p_go_interface}
    Sleep                          0.1s
    Write                          wps_pin any ${gc_pin}

    Check Gc Success

    # get gc interface
    ${p2p_gc_interface}=           sta1 get p2p interface
    Log                            ${p2p_gc_interface}

    Check Go Success

    Ping and Run Iperf             ${p2p_go_interface}               ${p2p_gc_interface}
