*** Keywords ***
Get Go MAC
    [Arguments]                    ${input}
    ${ret}=                        Get Lines Containing String       ${input}          go_dev_addr
    Log                            ${ret}
    ${ret}=                        Split String                      ${ret}            go_dev_addr=
    ${mac}=                        Get From List	             ${ret}            1
    RETURN                         ${mac}
    
Get P2P Interface
    [Arguments]                    ${host_alias}
    Switch Connection              ${host_alias}
    ${output}=                     Execute Command                   ${START_WPA_CLI_INTERFACE_COMMAND}
    Log                            ${output}
    ${p2p_interface}=              Get Lines Containing String       ${output}          p2p-wlan0-
    Log                            ${p2p_interface}
    Should Not Be Empty            ${p2p_interface}
    RETURN                         ${p2p_interface}

Ping and Run Iperf
    [Arguments]                    ${p2p_interface1}             ${p2p_interface2}

    Switch Connection              ${HOST1_BASH}
    Execute Command                sudo ifconfig ${p2p_interface1} ${HOST1_P2P_IP}
    ${output}=                     Execute Command                                       sudo ifconfig ${p2p_interface1}
    Log                            ${output}

    Switch Connection              ${HOST2_BASH}
    Execute Command                sudo ifconfig ${p2p_interface2} ${HOST2_P2P_IP}
    ${output}=                     Execute Command                                       sudo ifconfig ${p2p_interface2}
    Log                            ${output}
    ${ping_result}=                Execute Command                                       ping ${HOST1_P2P_IP} -I ${p2p_interface2} -c ${PING_COUNT}
    Log                            ${ping_result}
    Should Contain                 ${ping_result}                                        bytes from ${HOST1_P2P_IP} 

    Switch Connection              ${HOST1_BASH}
    Execute Command                sudo killall -9 iperf3 ; iperf3 -s -i 1 -p 9001 -D
    Switch Connection              ${HOST2_BASH}
    ${iperf_tx_result}=            Execute Command                                       iperf3 -t ${IPERF_RUN_TIME} -i 1 -c ${HOST1_P2P_IP} -p 9001 -P 4
    Should Contain                 ${iperf_tx_result}                                    connected

    Switch Connection              ${HOST1_BASH}
    Execute Command                sudo killall -9 iperf3 ; iperf3 -s -i 1 -p 9001 -D
    Switch Connection              ${HOST2_BASH}
    ${iperf_rx_result}=            Execute Command                                       iperf3 -t ${IPERF_RUN_TIME} -i 1 -c ${HOST1_P2P_IP} -p 9001 -P 4 -R
    Should Contain                 ${iperf_rx_result}                                    connected

    Log Many                       ${iperf_tx_result}                                    ${iperf_rx_result}

Start P2P Go
    [Arguments]                    ${start_go_command}
    Switch Connection              ${HOST1_WPA_CLI}
    Write                          interface p2p-dev-wlan0
    Sleep                          0.1s
    Write                          ${start_go_command}
    ${output}=                     Read                              delay=1s
    Log                            ${output}
    Should Contain                 ${output}                         OK
    ${p2p_go_mac}=                 Get Go MAC                        ${output}
    Log                            ${p2p_go_mac}
    RETURN                         ${p2p_go_mac}

P2P PBC Autonomous Go Enter Gc PIN In Go
    [Documentation]                Test steps:
    ...                            start autonomous go
    ...                            start wps pin on gc
    ...                            enter gc pin on go

    [Arguments]                    ${go_command}
    # start p2p go and get go mac
    ${p2p_go_mac}=                 Start P2P GO                      ${go_command}

    # start gc wps_pbc
    ${gc_pin}=                     Start P2P Gc PIN                  ${p2p_go_mac}

    # get go interface
    ${p2p_go_interface}=           Get P2P Interface                 ${HOST1_BASH}
    Log                            ${p2p_go_interface}

    # enter gc pin on go
    Switch Connection              ${HOST1_WPA_CLI}
    Write                          interface ${p2p_go_interface}
    Sleep                          0.1s
    Write                          wps_pin any ${gc_pin}

    Check Gc Success

    # get gc interface
    ${p2p_gc_interface}=           Get P2P Interface                 ${HOST2_BASH}
    Log                            ${p2p_gc_interface}

    Check Go Success

    Ping and Run Iperf             ${p2p_go_interface}               ${p2p_gc_interface}

Get P2p-dev-wlan0 Mac
    [Arguments]                    ${connection}
    Switch Connection              ${connection}
    ${output}=                     Execute Command                   ${P2P-DEV-WLAN0_STATUS_COMMAND}
    ${lines}                       Get Lines Containing String       ${output}            p2p_device_address
    ${mac}=                        Get Regexp Matches                ${lines}             ([0-9A-Fa-f]{2}[:-]){5}[0-9A-Fa-f]{2}
    Log                            ${mac}
    RETURN                         ${mac}[0]

   P2P Find Match Mac              ${HOST2_WPA_CLI}                  ${host1_p2p-dev-wlan0_mac}

   Switch Connection              ${HOST1_WPA_CLI}

   Write                          p2p_connect ${host1_p2p-dev-wlan0_mac} pbc go_intent=1

   Switch Connection              ${HOST1_WPA_CLI}
