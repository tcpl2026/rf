start 2g he20 ccmp sap
    [Arguments]                              ${channel}
    ssh login                                ${staut_host}

    remove conf file                         ${hostapd_conf}

    add content to hostapd conf file         "interface=${staut_sap_interface}"
    add content to hostapd conf file         "ctrl_interface=${hostapd_ctrl_interface}"
    add content to hostapd conf file         "driver=nl80211"

    add content to hostapd conf file         "ssid=${sap_ssid}"
    add content to hostapd conf file         "ignore_broadcast_ssid=0"

    add content to hostapd conf file         "hw_mode=g"
    add content to hostapd conf file         "channel=${channel}"

    add content to hostapd conf file         "wmm_enabled=1"

    add content to hostapd conf file         "ieee80211n=1"
    add content to hostapd conf file         "ht_capab=[SHORT-GI-20][LDPC]"

    add content to hostapd conf file         "ieee80211ax=1"
    add content to hostapd conf file         "he_bss_color=1"

    add content to hostapd conf file         "auth_algs=1"
    add content to hostapd conf file         "wpa=2"
    add content to hostapd conf file         "wpa_key_mgmt=WPA-PSK"
    add content to hostapd conf file         "wpa_pairwise=CCMP"
    add content to hostapd conf file         "wpa_passphrase=${sap_wpa2psk_passphrase}"
    add content to hostapd conf file         "wpa_group_rekey=${sap_wpa_rekey_interval}"

    ${output}=                               Execute Command                                   cat ${hostapd_conf}
    Log                                      ${output}

    kill process                             ${hostapd_conf}
    start sap                                ${hostapd_conf}
    Close Connection

start ccmp sta
    [Arguments]              ${host}               ${ssid}      ${sta_wlan_interface}
    ssh login                ${host}

    remove conf file         ${wpa_supplicant_conf}

    add content to wpa_supplicant conf file        "ctrl_interface=${wpa_supplicant_ctrl_interface}"
    add content to wpa_supplicant conf file        "network={"
    add content to wpa_supplicant conf file        'ssid="${ssid}"'
    add content to wpa_supplicant conf file        'psk="${sap_wpa2psk_passphrase}"'
    add content to wpa_supplicant conf file        "}" 

    ${output}=               Execute Command               cat ${wpa_supplicant_conf}
    Log                      ${output}

    kill process             ${wpa_supplicant_conf}
    start sta                ${sta_wlan_interface}               ${wpa_supplicant_conf}

    Close Connection

start iperf3 client
    [Arguments]             ${host}             ${bind_ip}     ${dst_ip}     ${dst_port}
    ssh login               ${host}
    Execute Command         sudo pkill -9 iperf3
    ${output}=              Execute Command     iperf3 -i 1 -P 10 -t ${iperf_run_time} -B ${bind_ip} -c ${dst_ip} -p ${dst_port}
    Log                     ${output}
    Should Contain          ${output}           connected

    Close Connection
